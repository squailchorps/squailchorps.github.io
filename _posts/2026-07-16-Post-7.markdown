using System.Collections.Generic;
using UnityEngine.UI;
using UnityEngine;
using Unity.VisualScripting;

public class game_logic : MonoBehaviour
{
    public GameObject secretCode;
    public List<button_logic> buttonList = new List<button_logic>() { };
    List<int> currentColorPositions = new List<int>();
    List<int> secretCodePull = new List<int>();
    public GameObject submit;
    public List<GameObject> colorIndicators = new List<GameObject>() { };

    // Start is called once before the first execution of Update after the MonoBehaviour is created
    void Start()
    {
        secretCodePull = secretCode.GetComponent<CodeGen>().secretCode;
        //Debug.Log("game_logic Start ran, secretCodePull count: " + secretCodePull.Count);
    }

    // Update is called once per frame
    void Update()
    {
        
    }

    public void checkColorAgainstSecretCode()

    {
        Debug.Log("buttonList.Count: " + buttonList.Count);
        Debug.Log("colorIndicators.Count: " + colorIndicators.Count);
        Debug.Log("secretCodePull.Count: " + secretCodePull.Count);
        currentColorPositions.Clear();
        for (int i = 0; i < buttonList.Count; i++)
        {
            
            currentColorPositions.Add(buttonList[i].colorPosition);

            if (currentColorPositions[i] == secretCodePull[i])
            {
                colorIndicators[i].GetComponent<Image>().color = Color.green;
            }
            else if (secretCodePull.Contains(currentColorPositions[i]) == true)
            {
                colorIndicators[i].GetComponent<Image>().color = Color.yellow;
            }
            else
            {
                colorIndicators[i].GetComponent<Image>().color = Color.red;
            }
        }
        //Add in the check against secretCode.
    } 
}
